## 1. 模块概述

### 1.1 职责边界

本模块负责：

- 市场状态自动机（四种状态+观望态的判定与切换）
- 四层漏斗筛选流程（粗筛→评分→Agent分析→加权投票）
- 策略一「周期底部量能异动」规则实现
- 策略二「趋势动量低吸」规则实现
- 回测引擎集成（vnpy BacktestingEngine）
- 开盘前决策复核
- 技术指标计算与回填

**不负责**：

- LLM Agent调度（见 DD-04）
- 下单执行（见 DD-05）
- 止损监控（见 DD-06）
- 新闻/数据采集（见 DD-02）

### 3.1 市场状态自动机

```
# 幂等性：否（状态切换有副作用）
function determine_market_state(trade_date: date) -> MarketState:
    # 1. 查询当前状态
    current_state = [DB] SELECT to_state FROM market_state_log
                        ORDER BY date DESC, created_at DESC LIMIT 1
    if current_state is None then
        current_state = MarketState.NO_MAINLINE  # 初始状态
    end if

    # 2. 检查冷却期
    last_switch_date = [DB] SELECT date FROM market_state_log
                           ORDER BY date DESC LIMIT 1
    cooldown_days = [CONFIG] "common.state_cooldown_days"  # 默认3天
    if (trade_date - last_switch_date).days < cooldown_days then
        [LOG] INFO "状态冷却期内，保持当前状态" current = current_state
              days_left = cooldown_days - (trade_date - last_switch_date).days
        return current_state
    end if

    # 3. 震荡保护检测
    recent_switches = [DB] SELECT * FROM market_state_log
                          WHERE date >= trade_date - timedelta(days=3)
                          ORDER BY date DESC
    oscillation_threshold = [CONFIG] "common.oscillation_threshold"
    if len(recent_switches) >= oscillation_threshold then
        # 来回切换过多，进入观望态
        new_state = MarketState.WATCH
        log_state_change(trade_date, current_state, new_state,
                        "震荡保护：{len(recent_switches)}天内切换{len(recent_switches)}次")
        return new_state
    end if

    # 4. 计算状态转换条件
    # 4.1 主线判定
    mainline_sectors = detect_mainline_sectors(trade_date)

    # 4.2 底部特征检测
    bottom_signals = detect_bottom_signals(trade_date)

    # 4.3 趋势特征检测
    trend_signals = detect_trend_signals(trade_date)

    # 5. 状态转换逻辑
    new_state = current_state  # 默认保持

    switch case current_state:
    case MarketState.NO_MAINLINE:
        if len(mainline_sectors) > 0 then
            new_state = MarketState.MAINLINE_CONFIRMED
        elif bottom_signals.detected then
            new_state = MarketState.BOTTOM_OPPORTUNITY
        end if

    case MarketState.MAINLINE_CONFIRMED:
        if trend_signals.confirmed then
            new_state = MarketState.TREND_UPTREND
        elif len(mainline_sectors) == 0 then
            new_state = MarketState.NO_MAINLINE
        end if

    case MarketState.TREND_UPTREND:
        if trend_signals.broken then
            new_state = MarketState.MAINLINE_CONFIRMED  # 趋势破坏，退回主线确认
            if len(mainline_sectors) == 0 then
                new_state = MarketState.NO_MAINLINE
            end if
        end if

    case MarketState.BOTTOM_OPPORTUNITY:
        if trend_signals.starting then
            new_state = MarketState.TREND_UPTREND  # 底部→趋势启动
        elif not bottom_signals.detected then
            new_state = MarketState.NO_MAINLINE
        end if

    case MarketState.WATCH:
        # 观望态：3天无切换后自动回到NO_MAINLINE
        if len(recent_switches) == 0 then
            new_state = MarketState.NO_MAINLINE
        end if
    end case

    # 6. 记录状态切换
    if new_state != current_state then
        reason = build_switch_reason(current_state, new_state, mainline_sectors,
                                      bottom_signals, trend_signals)
        log_state_change(trade_date, current_state, new_state, reason)

        # 发布状态变更事件
        [MQ] PUBLISH "market:state_change" = {
            from: current_state, to: new_state, reason: reason,
            main_line_sector: mainline_sectors[0] if mainline_sectors else None
        }
    end if

    return new_state
end function


# 主线板块检测
function detect_mainline_sectors(trade_date: date) -> list<SectorInfo>:
    capital_threshold = [CONFIG] "trend_momentum.sector_capital_ratio"  # 12%
    hot_days = [CONFIG] "trend_momentum.sector_hot_days"  # 2

    # 查询板块数据：资金集中度>=阈值 且 连续N天
    sectors = [DB] SELECT sector_code, sector_name, capital_ratio,
                   COUNT(*) as consecutive_days
               FROM sector_data
               WHERE trade_date >= trade_date - timedelta(days=hot_days)
                 AND capital_ratio >= capital_threshold
               GROUP BY sector_code, sector_name
               HAVING COUNT(*) >= hot_days
               ORDER BY capital_ratio DESC

    # 最多取2条主线（按资金集中度排序）
    max_mainlines = [CONFIG] "common.max_main_lines"
    return sectors[:max_mainlines]
end function


# 底部信号检测
function detect_bottom_signals(trade_date: date) -> BottomSignal:
    # 检查全市场是否有底部特征
    # 简化逻辑：是否有足够多的股票满足底部条件
    bottom_count = [DB] SELECT COUNT(DISTINCT stock_code)
                       FROM daily_klines k
                       WHERE k.trade_date >= trade_date - timedelta(days=60)
                       GROUP BY stock_code
                       HAVING (MAX(k.close) - MIN(k.close)) / MAX(k.close) >= 0.20
                          AND EXISTS (
                              SELECT 1 FROM daily_klines k2
                              WHERE k2.stock_code = k.stock_code
                                AND k2.trade_date >= trade_date - timedelta(days=5)
                                AND (k2.close - k2.open) / k2.open <= -0.05
                          )

    # 如果底部特征股票数量超过50只，认为有底部机会
    return BottomSignal(detected=(bottom_count >= 50), count=bottom_count)
end function


# 状态切换日志
function log_state_change(date: date, from_state: str, to_state: str, reason: str):
    [DB] INSERT INTO market_state_log (date, from_state, to_state, trigger_reason,
                                       main_line_sector, confidence)
    VALUES (date, from_state, to_state, reason, current_mainline, calculate_confidence())
    [LOG] INFO "市场状态切换" from = from_state to = to_state reason = reason
end function

function coarse_screen(trade_date: date, market_state: MarketState) -> list<CandidateStock>:
    candidates = []

    # 策略一：底部量能异动（仅在底部机会期或非主线状态执行）
    if market_state in [MarketState.BOTTOM_OPPORTUNITY, MarketState.NO_MAINLINE] then
        s1_candidates = screen_bottom_volume(trade_date)
        for c in s1_candidates do c.strategy_type = "bottom_volume" end for
        candidates.extend(s1_candidates)
    end if

    # 策略二：趋势动量低吸（仅在主线确认或趋势上升期执行）
    if market_state in [MarketState.MAINLINE_CONFIRMED, MarketState.TREND_UPTREND] then
        s2_candidates = screen_trend_momentum(trade_date)
        for c in s2_candidates do c.strategy_type = "trend_momentum" end for
        candidates.extend(s2_candidates)
    end if

    # 合并去重（策略一和策略二可能选出同一只股票）
    unique_candidates = deduplicate_by_stock_code(candidates)

    [LOG] INFO "粗筛完成" date = trade_date
           input = 5000 output = len(unique_candidates)
    return unique_candidates
end function


# 策略一粗筛
function screen_bottom_volume(trade_date: date) -> list[CandidateStock]:
    params = load_strategy_params("bottom_volume")

    # 1. 获取全市场符合条件的股票
    stocks = [DB] SELECT s.code, s.name, s.industry
               FROM stocks s
               WHERE s.is_st = FALSE
                 AND s.is_suspended = FALSE
                 AND (current_date - s.list_date).days >= params.min_list_days

    results = []
    for stock in stocks do
        # 2. 获取近60日K线
        klines = [DB] SELECT * FROM daily_klines
                     WHERE stock_code = stock.code
                       AND trade_date <= trade_date
                       AND adjust_flag = 'front'
                     ORDER BY trade_date DESC LIMIT 60

        if len(klines) < 60 then continue end if

        # 3. 条件1：近60日跌幅≥20%
        max_close = max([k.close for k in klines])
        latest_close = klines[0].close
        decline_pct = (max_close - latest_close) / max_close * 100
        if decline_pct < params.decline_60d_pct then continue end if

        # 4. 条件2：近5日内有单日跌幅≥5%
        recent_5d = klines[:5]
        has_sharp_drop = any([(k.close - k.open) / k.open * 100 <= -params.sharp_drop_5d_pct
                             for k in recent_5d])
        if not has_sharp_drop then continue end if

        # 5. 条件3：市值范围
        # 从fund_flows或基础信息获取市值（简化：用成交额估算）
        avg_amount = mean([k.amount for k in klines[:20]]) / 100000000  # 亿
        if avg_amount < params.min_daily_amount then continue end if

        # 6. 通过所有粗筛条件
        results.append(CandidateStock(
            stock_code = stock.code,
            stock_name = stock.name,
            strategy_type = "bottom_volume",
            decline_pct = decline_pct,
            avg_daily_amount = avg_amount
        ))
    end for

    return results
end function


# 策略二粗筛
function screen_trend_momentum(trade_date: date) -> list[CandidateStock]:
    params = load_strategy_params("trend_momentum")

    # 1. 获取当前主线板块成分股
    mainline_sectors = detect_mainline_sectors(trade_date)
    if len(mainline_sectors) == 0 then return [] end if

    sector_codes = [s.sector_code for s in mainline_sectors]
    # 获取板块成分股
    sector_stocks = [DB] SELECT DISTINCT stock_code FROM sector_constituents
                        WHERE sector_code IN sector_codes

    results = []
    for stock in sector_stocks do
        # 2. 获取K线数据
        klines = [DB] SELECT * FROM daily_klines
                     WHERE stock_code = stock.stock_code
                       AND trade_date <= trade_date
                       AND adjust_flag = 'front'
                     ORDER BY trade_date DESC LIMIT 60

        if len(klines) < 60 then continue end if

        # 3. 条件1：多头排列 MA5>MA10>MA20>MA60
        k = klines[0]  # 最新日
        if k.ma5 is None or k.ma10 is None or k.ma20 is None or k.ma60 is None then
            continue  # 指标未计算
        end if
        if not (k.ma5 > k.ma10 > k.ma20 > k.ma60) then continue end if

        # 4. 条件2：ADX ≥ 25
        if k.adx is None or k.adx < params.adx_threshold then continue end if

        # 5. 条件3：缩量回踩
        recent_3d = klines[:3]
        avg_5d_volume = mean([k2.volume for k2 in klines[:5]])
        is_shrinking = all([k2.volume < avg_5d_volume * params.volume_shrink_pct / 100
                           for k2 in recent_3d])
        if not is_shrinking then continue end if

        recent_3d_drop = (klines[0].close - klines[2].close) / klines[2].close * 100
        if abs(recent_3d_drop) > params.drop_3d_pct then continue end if

        # 6. 流动性检查
        avg_amount = mean([k2.amount for k2 in klines[:20]]) / 100000000
        if avg_amount < params.min_daily_amount then continue end if

        # 7. 通过
        results.append(CandidateStock(
            stock_code = stock.stock_code,
            strategy_type = "trend_momentum",
            adx = k.adx,
            avg_daily_amount = avg_amount
        ))
    end for

    return results
end function

### 3.3 四层漏斗筛选 — 第二层：多维度评分

```
# 幂等性：是
function score_candidates(candidates: list<CandidateStock>, trade_date: date)
    -> list[ScoredCandidate]:
    scored = []

    for candidate in candidates do
        # 获取所需数据
        klines = [DB] SELECT * FROM daily_klines
                     WHERE stock_code = candidate.stock_code
                       AND trade_date <= trade_date
                     ORDER BY trade_date DESC LIMIT 60
        fund_flow = [DB] SELECT * FROM fund_flows
                        WHERE stock_code = candidate.stock_code
                        ORDER BY trade_date DESC LIMIT 5
        news = [DB] SELECT * FROM news
                   WHERE related_stocks @> [candidate.stock_code]
                   ORDER BY published_at DESC LIMIT 10

        # 维度1：量价维度（20%）
        score_vp = calculate_volume_price_score(klines)

        # 维度2：资金维度（25%）
        score_fund = calculate_fund_score(fund_flow)

        # 维度3：情绪维度（15%）
        score_sent = calculate_sentiment_score(news, trade_date)

        # 维度4：主力行为维度（25%）
        score_mf = calculate_mainforce_score(klines, fund_flow)

        # 维度5：资本市场逻辑维度（15%）
        score_logic = calculate_capital_logic_score(candidate, news)

        # 加权总分
        total = score_vp * 0.20 + score_fund * 0.25 + score_sent * 0.15
              + score_mf * 0.25 + score_logic * 0.15

        scored.append(ScoredCandidate(
            **candidate,
            score_volume = score_vp,
            score_fund = score_fund,
            score_sentiment = score_sent,
            score_mainforce = score_mf,
            score_logic = score_logic,
            score_total = total
        ))
    end for

    # 按总分排序，取前15名
    scored.sort(key=lambda x: x.score_total, reverse=True)
    top_15 = scored[:15]

    # 评分截断检测
    if len(top_15) >= 6 then
        gap = top_15[4].score_total - top_15[5].score_total
        if gap >= 15 then
            [LOG] INFO "评分截断检测：前5名与第6名分差{gap:.1f}分"
        end if
    end if

    # 写入stock_pool（粗筛+评分结果）
    for s in scored do
        [DB] INSERT INTO stock_pool (date, stock_code, strategy_type, pass_coarse,
                score_total, score_volume, score_fund, score_sentiment,
                score_mainforce, score_logic)
        VALUES (trade_date, s.stock_code, s.strategy_type, TRUE,
                s.score_total, s.score_volume, s.score_fund, s.score_sentiment,
                s.score_mainforce, s.score_logic)
        ON CONFLICT (date, stock_code, strategy_type) DO UPDATE SET
            score_total = EXCLUDED.score_total, ...
    end for

    [LOG] INFO "评分完成" date = trade_date total = len(scored) top15 = len(top_15)
    return top_15
end function


# 主力行为维度评分
function calculate_mainforce_score(klines, fund_flow) -> float:

    latest = klines[0]

    if len(klines) >= 3 then

            days_after = [k for k in klines if k.trade_date > big_drop_day.trade_date][:2]
            for d in days_after do

            end for

    end if

        next_day = klines[0]

        elif next_day.volume > big_drop_day.volume then
            score -= 3  # 持续放量，倾向出货
        end if
    end if

    # 因子3：价格位置 — 已跌20%以上（+3分），已涨50%以上（-3分）
    max_close_60d = max([k.close for k in klines])
    min_close_60d = min([k.close for k in klines])
    position = (latest.close - min_close_60d) / (max_close_60d - min_close_60d)
    if position < 0.3 then score += 3     # 底部区域
    elif position > 0.8 then score -= 3   # 高位区域

    if len(fund_flow) > 0 then

        if ff.main_amount > 0 then score += 3  # 主力净流入
        elif ff.main_amount < 0 then score -= 3  # 主力净流出
        end if
    end if

    # 限制在0-100范围
    return max(0, min(100, score))

# 量价维度评分（20%权重）
# [PRD L234] 量价维度：均线排列完整性、量价关系、ADX趋势强度
function calculate_volume_price_score(klines) -> float:
    score = 50.0  # 基准分
    if len(klines) < 5 then return score end if
    latest = klines[0]

    # 因子1：均线排列完整性（±6分）
    # 多头排列：MA5>MA10>MA20>MA60 → +6分
    # 空头排列：MA5<MA10<MA20<MA60 → -6分
    # 部分排列 → 按满足比例加分
    if latest.ma5 is not None and latest.ma10 is not None
       and latest.ma20 is not None and latest.ma60 is not None then
        if latest.ma5 > latest.ma10 > latest.ma20 > latest.ma60 then
            score += 6  # 完美多头排列
        elif latest.ma5 < latest.ma10 < latest.ma20 < latest.ma60 then
            score -= 6  # 完美空头排列
        else
            # 部分排列：统计满足条件的相邻对数
            pairs = [(latest.ma5, latest.ma10), (latest.ma10, latest.ma20),
                     (latest.ma20, latest.ma60)]
            bullish_pairs = sum(1 for a, b in pairs if a > b)
            score += (bullish_pairs - 1.5) * 2  # 0对=-3, 1对=-1, 2对=+1, 3对=+3
        end if
    end if

    # 因子2：量价关系（±6分）
    # 放量上涨(+6)、缩量下跌(+3,健康回调)、放量下跌(-6,恐慌)、缩量上涨(-3,动力不足)
    if len(klines) >= 2 then
        prev = klines[1]
        price_change = (latest.close - prev.close) / prev.close
        vol_change = latest.volume / prev.volume if prev.volume > 0 else 1.0

        if price_change > 0.01 and vol_change > 1.3 then
            score += 6   # 放量上涨
        elif price_change > 0.01 and vol_change < 0.8 then
            score -= 3   # 缩量上涨，动力不足
        elif price_change < -0.01 and vol_change > 1.3 then
            score -= 6   # 放量下跌，恐慌
        elif price_change < -0.01 and vol_change < 0.8 then
            score += 3   # 缩量下跌，健康回调
        end if
    end if

    # 因子3：ADX趋势强度（±6分）
    # ADX>=25 强趋势 → +6分; ADX<20 无趋势 → -3分; 中间线性插值
    if latest.adx is not None then
        if latest.adx >= 25 then
            score += 6
        elif latest.adx < 20 then
            score -= 3
        else
            score += (latest.adx - 20) * 1.8  # 20→0, 25→9 取6
            score = min(score, 50.0 + 6)  # 上限
        end if
    end if

    # 因子4：价格位置（±3分）
    # 接近60日高点 → 趋势强但过热(0分); 接近60日低点 → 可能反转(+3分)
    if len(klines) >= 20 then
        max_close = max([k.close for k in klines[:60]])
        min_close = min([k.close for k in klines[:60]])
        range_val = max_close - min_close
        if range_val > 0 then
            position = (latest.close - min_close) / range_val
            if position < 0.3 then
                score += 3   # 底部区域，反转潜力
            elif position > 0.8 then
                score -= 2   # 高位区域，注意风险
            end if
        end if
    end if

    return max(0, min(100, score))
end function

# [PRD L235] 资金维度：主力资金净流入方向、CMF资金流向、大单占比
function calculate_fund_score(fund_flow) -> float:
    score = 50.0  # 基准分
    if len(fund_flow) == 0 then return score end if

    latest_ff = fund_flow[0]

    # 因子1：主力资金净流入方向（±8分）
    # 连续净流入>0 → 强势加分; 连续净流出 → 减分
    consecutive_inflow = 0
    consecutive_outflow = 0
    for ff in fund_flow do
        if ff.main_amount is not None then
            if ff.main_amount > 0 then
                consecutive_inflow += 1
                if consecutive_outflow > 0 then break end if
            elif ff.main_amount < 0 then
                consecutive_outflow += 1
                if consecutive_inflow > 0 then break end if
            end if
        end if
    end for

    if consecutive_inflow >= 3 then
        score += 8   # 连续3日以上主力净流入
    elif consecutive_inflow == 2 then
        score += 5
    elif consecutive_inflow == 1 then
        score += 2
    end if

    if consecutive_outflow >= 3 then
        score -= 8   # 连续3日以上主力净流出
    elif consecutive_outflow == 2 then
        score -= 5
    elif consecutive_outflow == 1 then
        score -= 2
    end if

    # 因子2：CMF资金流向（±6分）
    # CMF>0 资金流入; CMF<0 资金流出; CMF连续为正更可靠
    if latest_ff.cmf is not None then
        if latest_ff.cmf > 0 then
            score += 3  # 当日CMF为正
            # 检查是否连续为正
            if len(fund_flow) >= 3 then
                all_positive = all(ff.cmf is not None and ff.cmf > 0 for ff in fund_flow[:3])
                if all_positive then score += 3 end if  # 连续3日CMF为正
            end if
        elif latest_ff.cmf < 0 then
            score -= 3  # 当日CMF为负
            if len(fund_flow) >= 3 then
                all_negative = all(ff.cmf is not None and ff.cmf < 0 for ff in fund_flow[:3])
                if all_negative then score -= 3 end if  # 连续3日CMF为负
            end if
        end if
    end if

    # 因子3：大单占比（±6分）
    # 大单占比高 → 主力参与度高 → 加分
    if latest_ff.large_order_pct is not None then
        if latest_ff.large_order_pct > 0.30 then
            score += 6   # 大单占比>30%，主力高度参与
        elif latest_ff.large_order_pct > 0.20 then
            score += 3   # 大单占比>20%
        elif latest_ff.large_order_pct < 0.10 then
            score -= 3   # 大单占比<10%，主力参与度低
        end if
    end if

    # 因子4：净流入绝对值相对规模（±3分）
    # 主力净流入金额 / 当日成交额 的比值
    if latest_ff.main_amount is not None and latest_ff.net_amount is not None then
        ratio = abs(latest_ff.main_amount) / abs(latest_ff.net_amount) if latest_ff.net_amount != 0 else 0
        if ratio > 0.5 then
            score += 3  # 主力净流入占主导
        elif ratio < 0.2 then
            score -= 2  # 主力参与度微弱
        end if
    end if

    return max(0, min(100, score))
end function

    cold_sectors = [DB] SELECT COUNT(*) FROM sector_data

    # V1简化：检查相关新闻是否提及龙虎榜机构买入
    # 完整实现需接入龙虎榜数据（后续迭代）

    recent_news = [n for n in news if (trade_date - n.published_at).days <= 1]

    end if

function weighted_vote_and_risk_filter(scored_candidates: list[ScoredCandidate],

        weights = {main_hunter: 0.35, money_detective: 0.25,
                   mood_catcher: 0.15, experience_judge: 0.25}
    else:
        weights = {main_hunter: 0.20, money_detective: 0.25,
                   mood_catcher: 0.20, experience_judge: 0.35}
    end if

    # 2. 计算加权投票分数
    decisions = []
    for candidate in scored_candidates do
        agent_scores = candidate.agent_scores  # 来自DD-04 Agent分析

        # 验证Agent输出
        valid_scores = []
        for agent_name, score_data in agent_scores.items():
            if not AgentOutputValidator.validate(score_data) then
                [LOG] WARNING "Agent输出无效" agent = agent_name stock = candidate.stock_code
                continue
            end if
            valid_scores.append((agent_name, score_data.score, weights[agent_name]))
        end for

        # 处理不收敛情况
        if len(valid_scores) > 2 then
            score_values = [s[1] for s in valid_scores]
            score_range = max(score_values) - min(score_values)
            if score_range > 30 then
                # 丢弃最高分和最低分，取trimmed mean
                sorted_scores = sorted(valid_scores, key=lambda x: x[1])
                trimmed = sorted_scores[1:-1]
                final_score = mean([s[1] * s[2] for s in trimmed]) / mean([s[2] for s in trimmed])
                method = "trimmed_mean"
            else
                final_score = sum([s[1] * s[2] for s in valid_scores])
                method = "weighted_average"
            end if
        else
            final_score = candidate.score_total * 0.6  # Agent数据不足，回退到规则评分
            method = "fallback_rule_score"
        end if

        decisions.append(TradeDecision(
            stock_code = candidate.stock_code,
            strategy_type = candidate.strategy_type,
            final_score = final_score,
            agent_scores = agent_scores,
            method = method
        ))
    end for

    # 3. 排序取前3名
    decisions.sort(key=lambda x: x.final_score, reverse=True)
    top_3 = decisions[:3]

    # 4. 风控过滤
    approved = []
    for d in top_3 do
        # 流动性检查
        avg_amount = get_avg_daily_amount(d.stock_code, trade_date)
        min_amount = 1 if d.strategy_type == "bottom_volume" else 3
        if avg_amount < min_amount then
            d.reject_reason = "流动性不足"
            continue
        end if

        # 仓位检查
        current_position = get_current_position_pct(d.stock_code)
        max_position = [CONFIG] "common.max_position_per_stock"
        if current_position >= max_position then
            d.reject_reason = "仓位已达上限"
            continue
        end if

        # 因子拥挤度检查（简化实现）
        # TODO: 后续迭代

        # 通过风控
        d.final_decision = "buy"
        d.position_pct = calculate_position_size(d.final_score, market_state)
        d.stop_loss_pct = calculate_stop_loss(d.strategy_type, d.final_score)
        d.stop_profit_pct = calculate_stop_profit(d.strategy_type, d.final_score)
        approved.append(d)
    end for

    # 5. 更新stock_pool决策结果
    for d in decisions do
        [DB] UPDATE stock_pool SET
            final_decision = d.final_decision,
            final_score = d.final_score,
            agent_scores = d.agent_scores,

            stop_loss_pct = d.stop_loss_pct,
            stop_profit_pct = d.stop_profit_pct,
            reject_reason = d.reject_reason
        WHERE date = trade_date AND stock_code = d.stock_code
          AND strategy_type = d.strategy_type
    end for

    [LOG] INFO "加权投票完成" date = trade_date
           total = len(decisions) approved = len(approved)
    return approved
end function

function calculate_position_size(score: float, market_state: MarketState) -> float:
    # 基于评分和市场状态计算仓位
    base_pct = 10.0  # 默认10%

    if score >= 85 then base_pct = 25.0
    elif score >= 75 then base_pct = 20.0
    elif score >= 65 then base_pct = 15.0
    elif score >= 55 then base_pct = 10.0
    else base_pct = 5.0
    end if

    # 市场状态修正
    if market_state == MarketState.WATCH then
        base_pct *= 0.5  # 观望态减半
    elif market_state == MarketState.BOTTOM_OPPORTUNITY then
        base_pct *= 0.7  # 底部机会期降低仓位
    end if

    max_position = [CONFIG] "common.max_position_per_stock"
    return min(base_pct, max_position)
end function

    auction_data = get_auction_prices([d.stock_code for d in yesterday_decisions])

    # 5. 逐只复核
    results = []
    for decision in yesterday_decisions do
        review = ReviewResult(
            stock_code = decision.stock_code,
            original_decision = decision,
            status = "approved"  # 默认通过
        )

        # 5.1 集合竞价偏差检查
        auction = auction_data.get(decision.stock_code)
        if auction is not None then
            prev_close = get_prev_close(decision.stock_code)
            gap_pct = (auction.price - prev_close) / prev_close * 100

            if gap_pct >= gap_threshold then
                review.status = "rejected"
                review.reason = "高开{gap_pct:.1f}%，空间压缩"
            elif gap_pct <= -gap_threshold then
                review.status = "rejected"
                review.reason = "低开{gap_pct:.1f}%，逻辑可能改变"
            end if
        end if

        # 5.2 关联新闻检查
        stock_news = [n for n in premarket_news if decision.stock_code in n.related_stocks]
        negative_news = [n for n in stock_news if n.sentiment == "negative"]
        if len(negative_news) > 0 then
            review.status = "rejected"
            review.reason = "盘前重大利空: {negative_news[0].title[:50]}"
        end if

        results.append(review)
    end for

    approved_count = len([r for r in results if r.status == "approved"])
    rejected_count = len([r for r in results if r.status == "rejected"])

    [LOG] INFO "开盘前复核完成" approved = approved_count rejected = rejected_count

    # 如果全部取消，当日不开仓
    if approved_count == 0 then
        [LOG] WARNING "所有计划买入标的均被取消，当日不开仓"
    end if

    return results
end function

# vnpy回测适配层设计
# [PRD L2435] 提取 backtesting.py + template.py + object.py + constant.py + utility.py
#
# 核心矛盾：vnpy BacktestingEngine 设计为单标的CTA回测，

# 适配层1：FraxVerseStrategy — 将FraxVerse策略包装为vnpy AlphaStrategy
# [PRD L539] 策略类继承 AlphaStrategy
class FraxVerseStrategy(AlphaStrategy):
    # 每只标的使用独立的策略实例，但共享同一天的市场状态和粗筛结果
    parameters = {
        "market_state": None,       # 由外部注入当日市场状态
        "coarse_result": None,      # 由外部注入粗筛结果
        "score_data": None,         # 由外部注入评分数据
        "stop_loss_pct": 0.08,      # 默认8%止损
        "stop_profit_pct": 0.15,    # 默认15%止盈
    }

    function on_init():
        # 加载该标的的历史K线数据（vnpy自动完成）
        pass
    end function

    function on_bar(bar: BarData):
        trade_date = bar.datetime.date()

        # 检查当日粗筛+评分结果（由PortfolioBacktester注入）
        if self.parameters.coarse_result is None then return end if
        if bar.vt_symbol not in self.parameters.coarse_result then return end if

        score_data = self.parameters.score_data.get(bar.vt_symbol)
        if score_data is None then return end if

        # 开仓逻辑：根据评分和仓位比例决定
        if self.pos == 0 and score_data.final_decision == "buy" then
            # 计算开仓手数
            capital_per_stock = self.parameters.capital * score_data.position_pct / 100
            lots = int(capital_per_stock / (bar.close_price * 100))  # A股1手=100股
            if lots > 0 then
                self.buy(bar.close_price, lots)
            end if
        end if

        # 止损逻辑：硬约束，不经过Agent
        if self.pos > 0 then
            cost = self.pos_price
            loss_pct = (bar.close_price - cost) / cost
            if loss_pct <= -self.parameters.stop_loss_pct then
                self.sell(bar.close_price, abs(self.pos))  # 止损卖出
            elif loss_pct >= self.parameters.stop_profit_pct then
                self.sell(bar.close_price, abs(self.pos))  # 止盈卖出
            end if
        end if
    end function
end class


# 适配层2：PortfolioBacktester — 组合回测器
# 管理多标的回测，按日调度策略信号
class PortfolioBacktester:
    function __init__(strategy_type: str, start_date: date, end_date: date,
                       initial_capital: float, params: dict):
        self.strategy_type = strategy_type
        self.start_date = start_date
        self.end_date = end_date
        self.initial_capital = initial_capital
        self.params = params
        self.daily_portfolio = {}  # {date: {symbol: position}}
    end function

    function run() -> BacktestResult:
        # Step 1: 对回测期间每个交易日，运行完整的策略流程
        all_trade_dates = get_trade_dates(self.start_date, self.end_date)

        daily_equity = []  # 每日净值序列

        for trade_date in all_trade_dates do
            # 1.1 确定市场状态（复用策略引擎逻辑）
            market_state = determine_market_state(trade_date)

            # 1.2 粗筛（复用策略引擎逻辑）
            candidates = coarse_screen(trade_date, market_state)

            # 1.3 评分（复用策略引擎逻辑，不含Agent层）
            scored = score_candidates(candidates, trade_date)

            # 1.4 取消Agent层，回测中使用纯规则评分
            # [PRD L2539] P0阶段不依赖LLM，纯规则模式即可运行

            # 1.5 加权投票（回测模式下跳过Agent投票，直接用评分*0.6）
            for s in scored[:15] do
                s.final_score = s.score_total * 0.6  # 回测降权系数
                s.final_decision = "buy" if s.final_score >= 55 else "reject"
                s.position_pct = calculate_position_size(s.final_score, market_state)
                s.stop_loss_pct = 0.08
                s.stop_profit_pct = 0.15
            end for

            # 1.6 记录当日组合决策
            self.daily_portfolio[trade_date] = {
                s.stock_code: s for s in scored[:15] if s.final_decision == "buy"
            }
        end for

        # Step 2: 对每只出现在组合中的标的，运行vnpy单标的回测
        # 收集所有需要回测的标的
        all_symbols = set()
        for date_portfolio in self.daily_portfolio.values():
            all_symbols.update(date_portfolio.keys())

        symbol_backtest_results = {}  # {symbol: vnpy BacktestingResult}

        for symbol in all_symbols do
            # 获取该标的在回测期间的所有交易信号
            trade_signals = self._get_trade_signals(symbol)

            # 初始化vnpy单标的回测引擎
            engine = BacktestingEngine()
            engine.set_parameters(
                vt_symbol = symbol,  # 单标的，如 "600519.SH"
                interval = Interval.DAILY,
                start = self.start_date,
                end = self.end_date,
                rate = 0.0003,       # 交易费率万三 [PRD L2430]
                slippage = 0.01,     # 滑点
                size = 100,          # A股1手100股
                pricetick = 0.01,
                capital = self.initial_capital  # 虚拟全额，后续按position_pct折算
            )

            # 注入交易信号到策略实例
            strategy = FraxVerseStrategy(engine, self.params)
            strategy.parameters.trade_signals = trade_signals
            engine.add_strategy(strategy, self.params)

            engine.load_data()
            engine.run_backtesting()

            result = engine.calculate_result()
            symbol_backtest_results[symbol] = result
        end for

        # Step 3: 组合绩效聚合
        portfolio_result = self._aggregate_results(
            symbol_backtest_results, self.daily_portfolio
        )

        return portfolio_result
    end function

    # 获取某只标的在回测期间的交易信号
    function _get_trade_signals(symbol: str) -> list[TradeSignal]:
        signals = []
        for trade_date, portfolio in sorted(self.daily_portfolio.items()):
            if symbol in portfolio then
                s = portfolio[symbol]
                signals.append(TradeSignal(
                    date = trade_date,
                    action = "buy",
                    position_pct = s.position_pct,
                    stop_loss_pct = s.stop_loss_pct,
                    stop_profit_pct = s.stop_profit_pct
                ))
            else
                # 标的不在当日组合中 → 如果持仓则卖出
                signals.append(TradeSignal(
                    date = trade_date,
                    action = "exit_if_holding",
                    position_pct = 0
                ))
            end if
        end for
        return signals
    end function

    # 聚合多标的回测结果为组合绩效
    function _aggregate_results(symbol_results: dict, daily_portfolio: dict)
        -> BacktestResult:
        # 按日合并所有标的的持仓和盈亏
        # 计算组合级指标：年化收益、最大回撤、胜率、盈亏比
        daily_returns = {}
        for trade_date in sorted(daily_portfolio.keys()):
            day_pnl = 0
            for symbol, result in symbol_results.items():
                if trade_date in result.daily_results then
                    # 按position_pct折算盈亏
                    day_pnl += result.daily_results[trade_date].net_pnl
                end if
            end for
            daily_returns[trade_date] = day_pnl

        # 计算绩效指标
        total_return = sum(daily_returns.values()) / self.initial_capital
        max_drawdown = self._calc_max_drawdown(daily_returns)
        win_rate = self._calc_win_rate(daily_returns)
        profit_loss_ratio = self._calc_profit_loss_ratio(daily_returns)

        return BacktestResult(
            total_return = total_return,
            max_drawdown = max_drawdown,
            win_rate = win_rate,
            profit_loss_ratio = profit_loss_ratio,
            daily_equity = daily_returns,
            symbol_count = len(symbol_results)
        )
    end function
end class
