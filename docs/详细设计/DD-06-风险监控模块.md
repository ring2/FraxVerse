## 1. 模块概述

### 1.1 职责边界

本模块负责：

- **实时风控规则引擎** — 盘中持续监控回撤、胜率、盈亏比、连续亏损等指标，触发自动处置
- **舆情+风控联动** — 接收新闻情绪事件，自动执行仓位/交易限制
- **复苏路径管理** — 风控触发后的三路径复苏流程（A/B/C）
- **绩效分析** — 年化收益率、最大回撤、夏普比率、卡玛比率等指标计算
- **分层回溯验证** — 各层决策输出独立存储，交易结束后回溯验证
- **隔夜风险评估** — 持仓过夜前的风险检查
- **miniQMT健康监控** — 监控交易通道可用性

**不负责**：

- 止损条件绑定与触发执行（见 DD-05 止损监视器）
- 下单执行（见 DD-05）
- 新闻采集与情绪分析（见 DD-02 / DD-04）
- 策略参数调整建议（见 DD-07）

### 1.2 依赖关系

```
DD-06 风险监控模块
  ├── 依赖 DD-02 — daily_klines / news（行情与舆情数据）
  ├── 依赖 DD-03 — stock_pool / market_state_log（策略决策与市场状态）

  └── 被依赖 — DD-08前端（风控面板/告警推送）
```

```
function get_performance_analysis(request: PerformanceRequest) -> PerformanceResponse:
    end_date = today()
    start_date = calculate_start_date(request.period)

    results = {}
    for group_name, group_orders in groups do
        profits = [o.realized_pnl for o in group_orders if o.realized_pnl > 0]
        losses = [o.realized_pnl for o in group_orders if o.realized_pnl < 0]

            profit_loss_ratio: total_profit / len(profits) / (total_loss / len(losses))
                              if len(losses) > 0 and len(profits) > 0 else float('inf'),
            max_single_loss: min(losses) if losses else 0,
            total_trades: len(group_orders)
        }
    end for
```

## 4. 业务逻辑伪代码

### 4.1 盘中实时风控检查

```
# [PRD-T-158] 盘中风控响应延迟<=500ms  # PRD L2609 强约束
# 实现策略：内存缓存持仓+Redis实时价格，避免DB查询

# 幂等性：否（触发处置动作）
function intraday_risk_check():
    """
    盘中风控主循环，每1秒执行一次。
    所有持仓数据从内存缓存读取，价格从Redis读取，
    确保端到端延迟<=500ms。
    """
    # 前置检查：是否在交易时段
    if not is_trading_hours() then return end if

    # 前置检查：当前风控状态
    risk_status = [REDIS] GET "risk:status"
    if risk_status in ["paused", "stopped"] then return end if

    start_ts = now_milliseconds()

    # --- 1. 单日回撤检查 ---
    # [PRD-T-150] 单日回撤>=5%→清仓50%  # PRD L1085 强约束
    # [PRD-T-151] 单日回撤>=8%→清仓所有  # PRD L1086 强约束
    daily_drawdown = calculate_intraday_drawdown()  # 内存计算，不查DB

    if daily_drawdown >= 0.08 then
        trigger_drawdown_event(drawdown_pct=daily_drawdown, action="close_all")
        execute_close_all_positions()       # 清仓所有仓位
        [REDIS] SET "risk:status" = "paused"
        [REDIS] SET "risk:pause_until" = next_trade_date(3)  # 暂停3日
        [MQ] PUBLISH "system:alert" = {
            level: "critical",
            message: "单日回撤{daily_drawdown*100:.1f}%>=8%，清仓所有，暂停3日"
        }
    elif daily_drawdown >= 0.05 then
        trigger_drawdown_event(drawdown_pct=daily_drawdown, action="close_50pct")
        execute_close_half_positions()      # 清仓50%
        [REDIS] SET "risk:status" = "warning"
        [REDIS] SET "risk:position_cap" = "50"  # 仓位上限降至50%
        [MQ] PUBLISH "system:alert" = {
            level: "critical",
            message: "单日回撤{daily_drawdown*100:.1f}%>=5%，清仓50%"
        }
    end if

    # --- 2. 大盘单日跌幅检查 ---
    # [PRD-T-152] 大盘单日跌>=5%→清仓暂停3日  # PRD L1088 强约束
    market_change = get_market_realtime_change()  # Redis缓存的大盘实时涨跌
    if market_change is not None and market_change <= -0.05 then
        trigger_market_crash_event(change_pct=market_change)
        execute_close_all_positions()
        [REDIS] SET "risk:status" = "paused"
        [REDIS] SET "risk:pause_until" = next_trade_date(3)
        [MQ] PUBLISH "system:alert" = {
            level: "critical",
            message: "大盘单日跌{abs(market_change)*100:.1f}%>=5%，清仓暂停3日"
        }
    end if

    # --- 3. 盈亏比滚动监控 ---
    # [PRD-T-154] 盈亏比从1.5降到1.0以下→预警  # PRD L1164 强约束
    pl_ratio_rolling = get_rolling_pl_ratio()  # 内存计算
    if pl_ratio_rolling < 1.0 then
        trigger_pl_ratio_drop_event(pl_ratio=pl_ratio_rolling)
        [REDIS] SET "risk:status" = "warning"
        [MQ] PUBLISH "system:alert" = {
            level: "warning",
            message: "滚动盈亏比{pl_ratio_rolling:.2f}<1.0，发出预警"
        }
    end if

    # --- 4. 连续亏损监控 ---
    # [PRD-T-155] 连续亏损5次→自动暂停  # PRD L1165 强约束
    consecutive_losses = int([REDIS] GET "risk:consecutive_losses" or "0")
    if consecutive_losses >= 5 then
        trigger_consecutive_loss_event(count=consecutive_losses)
        [REDIS] SET "risk:status" = "paused"
        [MQ] PUBLISH "system:alert" = {
            level: "critical",
            message: "连续亏损{consecutive_losses}次，自动暂停交易"
        }
        # 进入复苏流程（见 4.4）
        initiate_recovery(event_type="consecutive_loss_5")
    end if

    # --- 5. miniQMT健康检查 ---
    # [PRD-T-156] miniQMT连续3次调用失败→停止下单+最高告警  # PRD L1166 强约束
    qmt_fail_count = int([REDIS] GET "risk:qmt_fail_count" or "0")
    if qmt_fail_count >= 3 then

            level: "critical",
            message: "miniQMT连续{qmt_fail_count}次调用失败，停止下单，最高告警"

    end if

    elapsed = now_milliseconds() - start_ts
    if elapsed > 500 then
        [LOG] WARNING "盘中风控检查超时" elapsed_ms = elapsed  # PRD L2609 强约束
    end if
end function


# 盘中回撤计算（纯内存，不查DB）
function calculate_intraday_drawdown() -> Decimal:
    """
    计算当日实时回撤率 = (开盘净值 - 当前净值) / 开盘净值
    所有数据从内存缓存读取，确保延迟<500ms
    """
    # 开盘净值（每日开盘时缓存到内存）
    opening_equity = memory_cache.get("opening_equity")
    if opening_equity is None then return Decimal(0) end if

    # 当前净值 = 现金 + 所有持仓市值
    current_equity = memory_cache.get("cash_balance")
    # [Schema: DD-05.positions] 持仓从内存缓存读取（DD-05止损监视器维护）
    positions = memory_cache.get("active_positions")  # list<Position>
    for pos in positions do
        # 实时价格从Redis读取（DD-02行情推送维护）
        current_price = [REDIS] GET "realtime_price:{pos.stock_code}"
        if current_price is not None then
            current_equity += pos.volume * Decimal(current_price)
        end if
    end for

    if opening_equity <= 0 then return Decimal(0) end if
    drawdown = (opening_equity - current_equity) / opening_equity
    return max(Decimal(0), drawdown)
end function
```

```
function initiate_recovery(event_type: str):
    """

# 判断复苏路径
function determine_recovery_path(agent_diagnosis: AgentDiagnosis,
                                  backtest_result: BacktestResult) -> str:
    """
    路径A — Agent分析出具体问题 + 回测证明新参数有效
    路径B — Agent分析不出原因，但回测无异常
    路径C — 回测也亏，或Agent认为市场不适合

    # 路径C判定：回测亏损 或 Agent认为市场不适合

        return "C"  # 回测也亏

        return "C"  # Agent认为市场不适合
    end if

    # 路径A判定：Agent分析出具体问题 + 回测证明新参数有效
    if agent_diagnosis.has_specific_issue and backtest_result.is_improved then
        return "A"  # 策略可修复
    end if

    # 路径B判定：Agent分析不出原因，但回测无异常
    if not agent_diagnosis.has_specific_issue and backtest_result.annual_return > 0 then
        return "B"  # 降级运行
    end if

    # 默认走路径C（最保守）
    return "C"
```

```
# 幂等性：否（生成风控事件）
# 每日14:50由APScheduler触发
function overnight_risk_assessment(trade_date: date):
    """
    收盘前评估隔夜风险，决定是否减仓过夜。
    """
    # [Schema: DD-05.positions] 只读引用
    positions = [DB] SELECT * FROM positions WHERE closed_at IS NULL

    if len(positions) == 0 then return end if  # 无持仓

    risk_score = 0  # 风险评分，0=安全，越高越危险
    reasons = []

    # 1. 外盘波动风险
    us_change = get_us_market_overnight_change()
    if us_change is not None then
        if abs(us_change) >= 0.03 then
            risk_score += 30
            reasons.append("外盘波动{us_change*100:.1f}%")
        elif abs(us_change) >= 0.02 then
            risk_score += 15
            reasons.append("外盘波动{us_change*100:.1f}%")
        end if
    end if

    # 2. 持仓集中度风险
    total_position_pct = calculate_total_position_pct(positions)
    if total_position_pct > 80 then
        risk_score += 20
        reasons.append("仓位{total_position_pct:.0f}%过高")
    end if

    # 3. 重大事件日历（财报季、FOMC等）
    upcoming_events = get_upcoming_risk_events()
    if len(upcoming_events) > 0 then
        risk_score += 15
        reasons.append("近期重大事件: {', '.join(upcoming_events[:3])}")
    end if

    # 4. 个股风险（是否有持仓股面临重大事件）
    for pos in positions do
        stock_events = get_stock_risk_events(pos.stock_code)
        if len(stock_events) > 0 then
            risk_score += 10
            reasons.append("{pos.stock_code}面临事件: {stock_events[0]}")
        end if
    end for

    # 5. 当日回撤情况
    daily_drawdown = calculate_intraday_drawdown()
    if daily_drawdown >= 0.03 then
        risk_score += 15
        reasons.append("当日回撤{daily_drawdown*100:.1f}%")
    end if

    # 处置决策
    if risk_score >= 60 then
        # 高风险：建议减仓至50%以下
        action = "reduce_overnight_position"
        target_pct = 30
```

```
function layer_backtrace_verification(trade_date: date):
    """
    交易结束后，回溯验证各层决策输出。
    模拟每层单独决策的结果，与最终决策对比。
    """
    # 获取当日所有决策
    # [Schema: DD-03.stock_pool] 只读引用
    pool_entries = [DB] SELECT * FROM stock_pool WHERE date = trade_date

    for entry in pool_entries do
        # 第一层：粗筛
        if entry.pass_coarse then
            coarse_decision = "buy"
        else
            coarse_decision = "reject"
        end if

        # 第二层：评分
        if entry.score_total >= 55 then
            score_decision = "buy"
        elif entry.score_total >= 40 then
            score_decision = "hold"
        else
            score_decision = "reject"
        end if

        # 第三层：Agent
        if entry.agent_scores is not None and len(entry.agent_scores) > 0:
            avg_agent_score = mean(entry.agent_scores.values())
            agent_decision = "buy" if avg_agent_score >= 60 else "reject"
        else
            agent_decision = "unknown"
        end if

        # 第四层：加权投票
        vote_decision = entry.final_decision

        # 实际执行
        # [Schema: DD-05.trade_orders] 只读引用
        executed = [DB] SELECT COUNT(*) FROM trade_orders
                        WHERE stock_code = entry.stock_code
                          AND trade_date = trade_date AND status = 'filled'
        final_decided = "buy" if executed > 0 else "no_action"

        # 写入分层回溯数据
        [DB] INSERT INTO layer_outputs (
            trade_date, layer_name, stock_code, output_data,
            would_decide, final_decided, is_consistent
        ) VALUES
        (trade_date, 'coarse', entry.stock_code,
         {pass: entry.pass_coarse, strategy: entry.strategy_type},
         coarse_decision, final_decided, coarse_decision == final_decided),
        (trade_date, 'score', entry.stock_code,
         {total: entry.score_total, volume: entry.score_volume, fund: entry.score_fund},
         score_decision, final_decided, score_decision == final_decided),
        (trade_date, 'agent', entry.stock_code,
         entry.agent_scores, agent_decision, final_decided,
         agent_decision == final_decided),
        (trade_date, 'vote', entry.stock_code,
         {final_score: entry.final_score, decision: entry.final_decision},
         vote_decision, final_decided, vote_decision == final_decided)
    end for
end function


# 月度贡献度报告
# 幂等性：是
function monthly_layer_contribution_report(year: int, month: int) -> LayerReport:
    """
    统计各层对最终决策的贡献度。
    识别噪声环节（一致性最低的层）。
    """
    start_date = date(year, month, 1)
    end_date = date(year, month + 1, 1) if month < 12 else date(year + 1, 1, 1)

    layer_outputs = [DB] SELECT * FROM layer_outputs
                         WHERE trade_date >= start_date AND trade_date < end_date

    layers = ['coarse', 'score', 'agent', 'vote']
    report = {}

    for layer in layers do
        layer_data = [lo for lo in layer_outputs if lo.layer_name == layer]
        total = len(layer_data)
        if total == 0 then continue end if

        consistent = len([lo for lo in layer_data if lo.is_consistent])
        consistency_rate = consistent / total

        # 噪声判定：一致性<50%
        is_noisy = consistency_rate < 0.50

        report[layer] = {
            total: total,
            consistent: consistent,
            consistency_rate: consistency_rate,
            is_noisy: is_noisy,
            would_buy_count: len([lo for lo in layer_data if lo.would_decide == "buy"]),
            actual_buy_count: len([lo for lo in layer_data if lo.final_decided == "buy"])
        }
    end for

    # 噪声环节自动精简建议
    noisy_layers = [l for l, r in report.items() if r.is_noisy]
    if len(noisy_layers) > 0 then
        [MQ] PUBLISH "system:alert" = {
            level: "info",
            message: "月度回溯: {', '.join(noisy_layers)}层一致性<50%，建议精简"
        }
    end if

    return report
end function
```

```
# 幂等性：否（生成风控事件）
function check_counterparty_risk(trade_date: date):
    """
    检查交易对手风险：
    1. 券商系统可用性（miniQMT连接状态）
    2. 流动性风险（个股成交额急剧萎缩）
    3. 涨跌停封板风险
    """
    # 1. 券商系统可用性 — 已由miniQMT健康监控覆盖（4.7）

    # 2. 流动性风险
    # [Schema: DD-05.positions] 只读引用
    positions = [DB] SELECT * FROM positions WHERE closed_at IS NULL

    for pos in positions do
        # 检查最近5日成交额趋势
        recent_amounts = [DB] SELECT amount FROM daily_klines
                              WHERE stock_code = pos.stock_code
                              ORDER BY trade_date DESC LIMIT 5
        # [Schema: DD-02.daily_klines] 只读引用

        if len(recent_amounts) >= 5 then
            avg_5d = mean(recent_amounts)
            avg_20d = [DB] SELECT AVG(amount) FROM daily_klines
                           WHERE stock_code = pos.stock_code
                             AND trade_date >= trade_date - timedelta(days=30)

            if avg_20d is not None and avg_5d < avg_20d * 0.30 then
                # 成交额萎缩至30%以下，流动性风险
                [DB] INSERT INTO risk_events (
                    event_type, event_level, trigger_value, threshold_value,
                    trigger_reason, action_taken, trade_date, is_intraday
                ) VALUES (
                    'counterparty_risk', 'warning',
                    avg_5d, avg_20d * 0.30,
                    '{pos.stock_code}流动性风险：5日均额/20日均额={avg_5d/avg_20d*100:.0f}%',
                    'reduce_position', trade_date, FALSE
                )
            end if
        end if

        # 3. 涨跌停封板风险
        latest_kline = [DB] SELECT * FROM daily_klines
                            WHERE stock_code = pos.stock_code
                            ORDER BY trade_date DESC LIMIT 1
        if latest_kline is not None then
            change_pct = (latest_kline.close - latest_kline.pre_close) / latest_kline.pre_close
            if change_pct >= 0.095 then
                # 涨停封板，卖出困难
                [LOG] WARNING "持仓{pos.stock_code}涨停封板，卖出受限"
            elif change_pct <= -0.095 then
                # 跌停封板，无法卖出
                [DB] INSERT INTO risk_events (
                    event_type, event_level, trigger_value, threshold_value,
                    trigger_reason, action_taken, trade_date, is_intraday
                ) VALUES (
                    'counterparty_risk', 'critical',
                    change_pct, -0.095,
                    '{pos.stock_code}跌停封板，无法卖出',
                    'mark_illiquid', trade_date, TRUE
                )
            end if
        end if
    end for
end function
```
