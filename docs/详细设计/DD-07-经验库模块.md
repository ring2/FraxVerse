## 1. 模块概述

### 1.1 职责边界

本模块负责：

- 经验的结构化归档（市场状态+操作+结果 → 记忆碎片）
- 经验库三阶段自适应匹配（冷启动/成长/成熟）
- 经验衰减与归档管理（6个月权重降低、3个月未验证归档）
- 自动调参建议生成 + 人工审批流程
- 参数变更审计日志（param_change_log）
- 调参后7天回测验证
- 回测经验预填充（source='backtest'）
- 冷启动期统计报告与规则自动提取
- 每周日反馈统计 → 门槛建议 → 微信推送审批
- 月度参数迭代流程

**不负责**：

- LLM Agent调度（见 DD-04）
- 交易执行（见 DD-05）
- 回测引擎执行（见 DD-03，本模块仅触发回测请求）
- 微信推送基础设施（见 DD-08，本模块仅构造推送内容）

### 1.2 设计原则

- [PRD-T-170] 不复用StockAgent记忆模块，全新设计 # PRD L2514 强约束
- [PRD-T-171] 自动调参改为"自动建议+人工审批"，系统不直接修改参数 # PRD L1631 强约束
- [PRD-T-176] 经验库三阶段自适应：冷启动/成长/成熟，模式自动切换 # PRD L1656 强约束

### 1.3 依赖关系


### 2.1 数据库表设计

#### 2.1.1 experiences 表 — 经验记录

```sql
CREATE TABLE experiences (
    id              BIGSERIAL       PRIMARY KEY,
    -- 场景描述
    market_state    VARCHAR(20)     NOT NULL,    -- 底部机会期/主线确认/趋势上升期/非主线状态/观望态
    sector          VARCHAR(32),                 -- 板块代码
    strategy_type   VARCHAR(20)     NOT NULL,    -- bottom_volume / trend_momentum
    stock_code      VARCHAR(10),                 -- 关联股票（可为空=规则型经验）
    -- 操作描述
    operation       VARCHAR(20)     NOT NULL,    -- buy / sell / hold / stop_loss / stop_profit
    operation_detail JSONB          DEFAULT '{}', -- 操作细节（入场价/出场价/仓位等）
    -- 结果描述
    result          VARCHAR(20)     NOT NULL,    -- profit / loss / break_even / skipped
    pnl_pct         NUMERIC(8,4),                -- 盈亏百分比
    holding_days    INTEGER,                     -- 持仓天数
    -- 评估
    score           NUMERIC(6,2)     NOT NULL,   -- 经验评分 0-100
    confidence      NUMERIC(4,2)     NOT NULL DEFAULT 50.0,  -- 置信度 0-100
    -- 标签与场景
    tags            JSONB           DEFAULT '[]',  -- 标签列表 ["量能异动","缩量回踩",...]
    scenario_hash   VARCHAR(64)     NOT NULL,    -- 场景特征哈希，用于去重
    -- 来源
    source          VARCHAR(20)     NOT NULL DEFAULT 'real',  -- backtest / real / rule_extract
    source_id       BIGINT,                     -- 来源关联ID（如回测结果ID）
    -- 经验衰减
    is_archived     BOOLEAN         NOT NULL DEFAULT FALSE,  -- 是否已归档
    last_verified_at TIMESTAMPTZ,               -- 最近一次验证时间
    weight          NUMERIC(4,2)     NOT NULL DEFAULT 100.0, -- 当前权重（衰减后降低）
    -- 关联
    related_trade_id BIGINT,                    -- 关联交易订单ID
    feedback_text   TEXT,                        -- 反馈描述（来自feedback.log）
    -- 通用字段
    created_at      TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
    updated_at      TIMESTAMPTZ     NOT NULL DEFAULT NOW(),

    -- 唯一约束：同一场景哈希不重复存储
    UNIQUE(scenario_hash)
);

-- 高频查询索引
CREATE INDEX idx_experiences_market_state ON experiences(market_state, created_at DESC);
CREATE INDEX idx_experiences_strategy ON experiences(strategy_type, created_at DESC);
CREATE INDEX idx_experiences_sector ON experiences(sector, created_at DESC);
CREATE INDEX idx_experiences_source ON experiences(source, created_at DESC);
CREATE INDEX idx_experiences_score ON experiences(score DESC) WHERE is_archived = FALSE;
CREATE INDEX idx_experiences_scenario ON experiences(scenario_hash) WHERE is_archived = FALSE;
-- 未归档+未衰减的经验（匹配查询主路径）
CREATE INDEX idx_experiences_active ON experiences(market_state, strategy_type, is_archived)
    WHERE is_archived = FALSE AND weight > 0;

COMMENT ON TABLE experiences IS '经验记录，系统自我进化的核心知识库';
COMMENT ON COLUMN experiences.scenario_hash IS '场景特征哈希=hash(market_state+sector+strategy_type+operation+tags)，用于去重';
COMMENT ON COLUMN experiences.source IS 'backtest=回测预填充 / real=实盘经验 / rule_extract=统计规则自动提取';
COMMENT ON COLUMN experiences.weight IS '经验权重，超过6个月降低，3个月未验证归档后设为0';
```

```
function match_experience(req: MatchRequest) -> MatchResult:
    # 1. 判定当前阶段
    total_count = [DB] SELECT COUNT(*) FROM experiences WHERE is_archived = FALSE
    phase = determine_phase(total_count)  # T-177: 冷启动阈值100条

    # 2. 根据阶段选择匹配策略
    switch case phase:
    case "cold_start":
        # 冷启动期：只出统计数据，不做单条匹配
        stats = generate_cold_start_statistics(req.market_state, req.strategy_type)
        return MatchResult(phase="cold_start", statistics=stats, matches=[])

    case "growth":
        # 成长期：统计为主+匹配为辅
        stats = generate_cold_start_statistics(req.market_state, req.strategy_type)
        matches = similarity_match(req, weight_factor=0.3)  # 匹配权重30%
        return MatchResult(phase="growth", statistics=stats, matches=matches)

    case "mature":
        # 成熟期：匹配为主+统计为辅
        matches = similarity_match(req, weight_factor=0.8)  # 匹配权重80%
        stats = generate_cold_start_statistics(req.market_state, req.strategy_type)
        return MatchResult(phase="mature", statistics=stats, matches=matches)
    end case
end function
```

```
# [PRD-T-171] 自动调参改为"自动建议+人工审批" # PRD L1631 强约束
# [PRD-T-174] 同一参数7天内只能调整1次 # PRD L1641 强约束
function suggest_param_change(req: ParamSuggestRequest) -> ParamChangeLog:
    # 1. 检查7天内是否已有该参数的变更记录
    recent_change = [DB] SELECT * FROM param_change_log
                    WHERE strategy_type = req.strategy_type
                      AND param_key = req.param_key
                      AND status IN ('pending', 'approved')
                      AND created_at >= NOW() - INTERVAL '7 days'
                    ORDER BY created_at DESC LIMIT 1

    if recent_change is not None then
        [RAISE] Error(40710, "该参数7天内已有变更记录，不可重复调整")  # PRD L1641 强约束
    end if

    # 2. 获取当前参数值
    # [Schema: DD-03.strategy_params] 策略参数表
    current_param = [DB] SELECT param_value FROM strategy_params
                     WHERE strategy_type = req.strategy_type
                       AND param_key = req.param_key
    if current_param is None then
        [RAISE] Error(40702, "参数不存在")
    end if

    # 3. 计算建议新值
    suggested_value = calculate_suggested_value(
        req.strategy_type, req.param_key, current_param.param_value,
        req.suggestion_type, req.reason
    )

    # 4. 记录变更日志 — 状态为pending
```

```
# 冷启动期统计规则自动提取
function extract_rules_from_statistics(by_strategy, by_market_state, by_sector)
    -> list[ExtractedRule]:
    rules = []

    # 规则1：胜率最高的板块+策略组合 → "偏好规则"
    for entry in by_sector do
        for s in by_strategy do
            combination = [DB] SELECT
                COUNT(*) as total,
                SUM(CASE WHEN result = 'profit' THEN 1 ELSE 0 END) / COUNT(*)::float as win_rate
            FROM experiences
            WHERE sector = entry.sector
              AND strategy_type = s.strategy_type
              AND is_archived = FALSE

            if combination.total >= 3 and combination.win_rate >= 0.70 then
                rules.append(ExtractedRule(
                    rule_type = "prefer",
                    description = f"{entry.sector}+{s.strategy_type}：胜率{combination.win_rate:.0%}",
                    confidence = min(combination.win_rate * 100, 80.0)
                ))
            end if
        end for
    end for

    # 规则2：亏损最多的市场状态 → "回避规则"
    for ms in by_market_state do
        if ms.total >= 3 and ms.win_rate < 0.40 then
            rules.append(ExtractedRule(
                rule_type = "avoid",
                description = f"{ms.market_state}状态：胜率仅{ms.win_rate:.0%}",
                confidence = min((1 - ms.win_rate) * 100, 80.0)
            ))
        end if
    end for

    # 将提取的规则写入经验库（source='rule_extract'）
    for rule in rules do
        scenario_hash = compute_scenario_hash(
            rule.rule_type, '', '', 'rule', [rule.description]

        existing = [DB] SELECT id FROM experiences WHERE scenario_hash = scenario_hash
        if existing is None then
            [DB] INSERT INTO experiences (
                market_state, strategy_type, operation, result,

                'any', 'any', 'rule', rule.rule_type,

# 成熟期相似度匹配
function similarity_match(req: MatchRequest, weight_factor: float)
    -> list[MatchedExperience]:
    # 1. 查询候选经验
    candidates = [DB] SELECT * FROM experiences
                  WHERE is_archived = FALSE
                    AND weight > 0
                    AND (market_state = req.market_state OR market_state = 'any')
                    AND (strategy_type = req.strategy_type OR strategy_type = 'any')
                  ORDER BY score DESC LIMIT 50

    if len(candidates) == 0 then return [] end if

    # 2. 计算相似度
    scored_matches = []
    for exp in candidates do
        similarity = 0.0

        # 市场状态完全匹配 → +30%
        if exp.market_state == req.market_state then
            similarity += 0.30
        end if

        # 策略类型完全匹配 → +20%
        if exp.strategy_type == req.strategy_type then
            similarity += 0.20
        end if

        # 板块匹配 → +20%
        if req.sector is not None and exp.sector == req.sector then
            similarity += 0.20
        end if

        # 标签重叠度 → +30%
        if len(req.tags) > 0 and len(exp.tags) > 0 then
            overlap = len(set(req.tags) & set(exp.tags))
            total = len(set(req.tags) | set(exp.tags))
            tag_similarity = overlap / total if total > 0 else 0
            similarity += tag_similarity * 0.30
        end if

        # 应用权重衰减
        decayed_weight = exp.weight / 100.0  # 衰减后权重
        final_score = similarity * weight_factor * decayed_weight

        scored_matches.append(MatchedExperience(
            experience = exp,
            similarity = similarity,
            final_score = final_score
        ))
    end for

    # 3. 排序取Top10
    scored_matches.sort(key=lambda x: x.final_score, reverse=True)
    return scored_matches[:10]
end function
```

```
            [LOG] INFO "参数7天内已变更，跳过" param = param.param_key
            continue
        end if

        # 3.2 分析该参数对胜率的影响
        analysis = analyze_param_impact(param, last_month_experiences)

        # 3.3 如果分析表明参数需要调整，生成建议
        if analysis.needs_adjustment then
            try:
                suggest_param_change(ParamSuggestRequest(
                    suggestion_type = "monthly_iteration",
                    strategy_type = param.strategy_type,
                    param_key = param.param_key,
                    reason = f"月度迭代分析：{analysis.reason}"
                ))
            except Error as e:
                [LOG] WARNING "月度参数建议生成失败" param = param.param_key error = e
            end try
        end if
    end for

    [LOG] INFO "=== 月度参数迭代完成 ==="
end function


# 分析参数对胜率的影响
function analyze_param_impact(param: StrategyParam, experiences: list)
    -> ParamAnalysis:
    # 根据参数类型选择分析策略
    result = ParamAnalysis(needs_adjustment=False)

    # 阈值类参数：检查当前阈值下的胜率
    if param.param_type == 'int' and 'pct' in param.param_key then
        threshold = int(param.param_value)

        # 经验中与该阈值相关的操作
        related = [e for e in experiences if e.strategy_type == param.strategy_type]

        if len(related) >= 5 then
            win_rate = sum(1 for e in related if e.result == 'profit') / len(related)

            # 胜率低于50% → 可能需要调整阈值
            if win_rate < 0.50 then
                result.needs_adjustment = True
                result.reason = f"上月胜率{win_rate:.0%}低于50%，建议调整阈值"
                result.suggested_direction = "raise" if "当前阈值过低导致误入" in "" else "lower"
            end if

            # 胜率高于80%但交易次数少 → 阈值可能过高
            if win_rate > 0.80 and len(related) < 5 then
                result.needs_adjustment = True
                result.reason = f"胜率高但交易仅{len(related)}次，阈值可能过高"
                result.suggested_direction = "lower"
            end if
        end if
    end if

    return result
end function
```

```
                AND created_at < NOW() - INTERVAL '6 months'

        # 计算衰减后权重：每超过6个月1天，权重降低0.5%
        months_overdue = (NOW() - exp.created_at).days / 30 - 6
        new_weight = max(0.0, 100.0 - months_overdue * 5.0)

        [DB] UPDATE experiences SET
            weight = new_weight,
            updated_at = NOW()
        WHERE id = exp.id
    end for

    [LOG] INFO "经验衰减完成" count = len(decayed)

    # 2. 经验归档：3个月以上未验证的旧经验归档

    archived = [DB] SELECT * FROM experiences

    for exp in archived do
```

```
function get_experience_insight(market_state: str, strategy_type: str,

    )

    # 4. 生成推荐
    if phase == "cold_start" then

        if match_result.statistics is not None then

    elif phase == "growth" then

        if len(match_result.matches) > 0 then

                f"{top.experience.operation} → {top.experience.result}，"

            )

    else  # mature
        # 成熟期：匹配为主
        if len(match_result.matches) >= 3 then
```

```
# 计算参数建议新值
function calculate_suggested_value(strategy_type: str, param_key: str,
                                    current_value: str,
                                    suggestion_type: str,
                                    reason: str) -> str:
    current_num = float(current_value)

    # 根据建议类型和方向调整
    if suggestion_type == "threshold_adjust" then
        if "过严" in reason or "错过" in reason or "lower" in reason then
            # 降门槛：减少5-15%
            suggested = current_num * 0.90  # 默认降低10%
            # 最小步长：整数参数至少变化1
            if current_num == int(current_num) then
                suggested = max(1, current_num - 1)
            end if
        elif "过松" in reason or "误入" in reason or "raise" in reason then
            # 升门槛：增加5-15%
            suggested = current_num * 1.10  # 默认提升10%
            if current_num == int(current_num) then
                suggested = current_num + 1
            end if
        else
            suggested = current_num  # 不变
        end if
    elif suggestion_type == "monthly_iteration" then
        # 月度迭代：更保守的调整幅度
        suggested = current_num * 1.05  # 默认5%
    else
        suggested = current_num
    end if

    # 限制调整幅度：不超过当前值的20%
    max_change = abs(current_num * 0.20)
    if abs(suggested - current_num) > max_change then
        if suggested > current_num then
            suggested = current_num + max_change
        else
            suggested = current_num - max_change
        end if
    end if

    # 类型还原
    if current_value == str(int(current_num)) then
        return str(int(round(suggested)))
    end if

    return f"{suggested:.2f}"
end function
```
