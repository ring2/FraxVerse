## 1. 模块概述

### 1.1 职责边界

本模块负责：

- 前端工程架构与目录结构设计（从零搭建，不复用 QuantMind）[PRD-T-190] # PRD L2467 强约束
- FraxVerse 宇宙视觉系统的组件化实现（深空背景、粒子特效、碎片卡片、星图）[PRD-T-198][PRD-T-200]
- 品牌色彩系统与设计语言的工程化落地 [PRD-T-199]
- 全部页面组件与路由体系（PC端12页 + 移动端核心4页）[PRD-T-201]
- Zustand 状态管理层 [PRD-T-195]
- API 调用封装、WebSocket 实时推送、认证前端实现
- 错误码映射与用户友好提示
- 页面加载性能保障（≤2秒）[PRD-T-202]

**不负责**：

- 后端 API 逻辑（各后端 DD 负责）
- 数据库表结构设计（各后端 DD 负责）
- LLM 调用与 Agent 编排（DD-04 负责）
- 交易执行逻辑（DD-05 负责）

### 2.2 Zustand Store 模型定义

#### 2.2.1 useAuthStore — 认证状态

```typescript
// [PRD-T-191] React 18 + TypeScript + Vite # PRD L1781 强约束
// [PRD-T-195] Zustand 状态管理，不得使用 Redux # PRD L1786 强约束

interface AuthState {
  // 状态
  isAuthenticated: boolean;
  user: { id: number; username: string } | null;
  accessToken: string | null;      // 仅内存，不持久化
  refreshToken: string | null;     // localStorage/sessionStorage
  isLoading: boolean;
  error: string | null;

  // 动作
  login: (username: string, password: string, rememberMe: boolean) => Promise<void>;
  init: (payload: InitRequest) => Promise<void>;
  logout: () => Promise<void>;
  refreshAccessToken: () => Promise<boolean>;
  clearError: () => void;
  setTokens: (accessToken: string, refreshToken: string, rememberMe: boolean) => void;
}

// 竞态锁：全局 Promise 引用，防止并发 refresh
let refreshPromise: Promise<boolean> | null = null;
```

#### 2.2.2 useMarketStore — 行情状态

```typescript
interface MarketState {
  // 大盘概览
  marketState: 'bull' | 'bear' | 'neutral' | null;
  marketSummary: {
    shIndex: number;        // 上证指数
    szIndex: number;        // 深证成指
    cybIndex: number;       // 创业板指
    shChangePct: number;
    szChangePct: number;
    cybChangePct: number;
    turnover: number;       // 成交额(亿)
    advanceCount: number;   // 上涨家数
    declineCount: number;   // 下跌家数
  } | null;

  // K线数据
  klineData: Map<string, {        // key: stockCode
    daily: KlineItem[];
    realtime: RealtimeQuote | null;
    signals: SignalItem[];
  }>;

  // 资金流向
  fundFlow: FundFlowItem[] | null;

  // 加载状态
  isLoadingSummary: boolean;
  isLoadingKline: Map<string, boolean>;

  // 动作
  fetchMarketSummary: () => Promise<void>;
  fetchKline: (stockCode: string, period: string) => Promise<void>;
  fetchRealtimeQuote: (stockCode: string) => Promise<void>;
  updateRealtimeQuote: (quote: RealtimeQuote) => void;  // WS推送
  updateMarketState: (state: string) => void;            // WS推送
}
```

```typescript
// services/api.ts

import axios, { AxiosError, InternalAxiosRequestConfig } from 'axios';
import { getAccessToken, getRefreshToken, setTokens, clearTokens } from '@/auth/tokenManager';

// [PRD-T-191] React 18 + TypeScript + Vite # PRD L1781 强约束

const api = axios.create({
  baseURL: '/api/v1',
  timeout: 30000,
  headers: { 'Content-Type': 'application/json' },
});

// ========== 请求拦截器：注入Access Token ==========
api.interceptors.request.use((config: InternalAxiosRequestConfig) => {
  const token = getAccessToken();    // 从内存读取
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// ========== 响应拦截器：401自动续期 + 错误码映射 ==========
api.interceptors.response.use(
  (response) => response,
  async (error: AxiosError<ApiErrorResponse>) => {
    const originalRequest = error.config as InternalAxiosRequestConfig & { _retry?: boolean };

    if (error.response?.status === 401 && !originalRequest._retry) {
      originalRequest._retry = true;
      const refreshed = await tryRefreshToken();   // 全局Promise锁
      if (refreshed) {
        originalRequest.headers.Authorization = `Bearer ${getAccessToken()}`;
        return api.request(originalRequest);       // 重发原请求
      } else {
        clearTokens();
        // React Router v6 navigate
        window.location.href = '/login';
        return Promise.reject(error);
      }
    }

    // 错误码映射为用户友好消息
    if (error.response?.data?.code) {
      const friendlyMsg = mapErrorCodeToMessage(error.response.data.code);
      error.message = friendlyMsg;
    }

    return Promise.reject(error);
  }
);
```

```typescript
// auth/tokenManager.ts

// [PRD-T-191] React 18 + TypeScript # PRD L1781 强约束
// Access Token 仅存内存，Refresh Token 存 localStorage

let accessToken: string | null = null;
let refreshPromise: Promise<boolean> | null = null;   // 全局竞态锁

export function getAccessToken(): string | null {
  return accessToken;
}

export function getRefreshToken(): string | null {
  return sessionStorage.getItem('refresh_token')
    || localStorage.getItem('refresh_token');
}

export function setTokens(access: string, refresh: string, rememberMe: boolean): void {
  accessToken = access;
  if (rememberMe) {
    localStorage.setItem('refresh_token', refresh);
  } else {
    sessionStorage.setItem('refresh_token', refresh);
  }
}

export function clearTokens(): void {
  accessToken = null;
  sessionStorage.removeItem('refresh_token');
  localStorage.removeItem('refresh_token');
}

// Token续期 — 全局Promise锁防竞态
// [PRD L1467] 多请求同时401时仅发起1次refresh
export async function tryRefreshToken(): Promise<boolean> {
  // 已有进行中的refresh请求，复用之
  if (refreshPromise !== null) {
    return refreshPromise;
  }

  const refreshToken = getRefreshToken();
  if (!refreshToken) {
    return false;
  }

  // 创建全局Promise锁
  refreshPromise = (async () => {
    try {
      const response = await axios.post('/api/v1/auth/refresh', {
        refreshToken,
      });
      const { accessToken: newAccess, refreshToken: newRefresh } = response.data.data;
      setTokens(newAccess, newRefresh, true);
      return true;
    } catch {
      clearTokens();
      return false;
    } finally {
      refreshPromise = null;      // 释放锁
    }
  })();

  return refreshPromise;
}
```

```typescript
// components/cosmos/ParticleCanvas.tsx

// [PRD-T-198] 宇宙深空背景+粒子滑动特效 # PRD L1266 强约束
// [PRD-T-200] 粒子特效参数(100-150个/0.3-1.5px等) # PRD L1285 强约束

import { useRef, useEffect } from 'react';
import { PARTICLE_CONFIG } from '@/theme/particles';

interface Particle {
  x: number;
  y: number;
  vx: number;
  vy: number;
  size: number;
  opacity: number;
}

export function ParticleCanvas(): JSX.Element {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const particlesRef = useRef<Particle[]>([]);
  const animationRef = useRef<number>(0);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    const config = PARTICLE_CONFIG;
    // 移动端减少粒子
    const isMobile = window.innerWidth < 768;
    const particleCount = isMobile ? config.mobileCount : config.count;

    // 初始化粒子
    function initParticles(): Particle[] {
      const particles: Particle[] = [];
      for (let i = 0; i < particleCount; i++) {
        particles.push({
          x: Math.random() * canvas!.width,
          y: Math.random() * canvas!.height,
          vx: (Math.random() - 0.5) * config.speed * 2,
          vy: (Math.random() - 0.5) * config.speed * 2,
          size: config.minSize + Math.random() * (config.maxSize - config.minSize),
          opacity: config.minOpacity + Math.random() * (config.maxOpacity - config.minOpacity),
        });
      }
      return particles;
    }

    // 绘制连线
    function drawLinks(particles: Particle[]): void {
      for (let i = 0; i < particles.length; i++) {
        for (let j = i + 1; j < particles.length; j++) {
          const dx = particles[i].x - particles[j].x;
          const dy = particles[i].y - particles[j].y;
          const dist = Math.sqrt(dx * dx + dy * dy);
          if (dist < config.linkDistance) {
            ctx!.strokeStyle = config.linkColor;
            ctx!.lineWidth = config.linkWidth;
            ctx!.beginPath();
            ctx!.moveTo(particles[i].x, particles[i].y);
            ctx!.lineTo(particles[j].x, particles[j].y);
            ctx!.stroke();
          }
        }
      }
    }

    // 动画帧
    function animate(): void {
      if (!ctx || !canvas) return;
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      // 更新粒子位置
      for (const p of particlesRef.current) {
        p.x += p.vx;
        p.y += p.vy;

        // 边界反弹
        if (p.x < 0 || p.x > canvas.width) p.vx *= -1;
        if (p.y < 0 || p.y > canvas.height) p.vy *= -1;
      }

      // 绘制连线
      drawLinks(particlesRef.current);

      // 绘制粒子
      for (const p of particlesRef.current) {
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
        ctx.fillStyle = `rgba(127, 119, 221, ${p.opacity})`;  // 星云紫系
        ctx.fill();
      }

      animationRef.current = requestAnimationFrame(animate);
    }

    // 尺寸自适应
    function resize(): void {
      canvas!.width = window.innerWidth;
      canvas!.height = window.innerHeight;
    }

    resize();
    particlesRef.current = initParticles();
    animate();

    window.addEventListener('resize', resize);

    return () => {
      cancelAnimationFrame(animationRef.current);
      window.removeEventListener('resize', resize);
    };
  }, []);

  return (
    <canvas
      ref={canvasRef}
      style={{
        position: 'fixed',
        top: 0,
        left: 0,
        width: '100%',
        height: '100%',
        zIndex: 0,
        pointerEvents: 'none',
        background: PARTICLE_CONFIG.background,     // #06060f 宇宙深空
      }}
    />
  );
}
```

```typescript
import { useNavigate } from 'react-router-dom';
import { Steps, Form, Input, Select, Button, Result, message, Modal } from 'antd';

import { BrandLogo } from '@/components/cosmos/BrandLogo';
import { FRAX_COLORS } from '@/theme/colors';

type SetupStep = 0 | 1 | 2;

export default function SetupPage() {
  const navigate = useNavigate();
  const [currentStep, setCurrentStep] = useState<SetupStep>(0);
  const [isCompleted, setIsCompleted] = useState(false);
  const [submitting, setSubmitting] = useState(false);

  // Step 1: 创建管理员
  const [adminForm] = Form.useForm();
  // Step 2: 核心配置
  const [configForm] = Form.useForm();
  // Step 3: 预览汇总
  const [summary, setSummary] = useState<{
    username: string;
    deepseekKey: string;
    tradeMode: string;
    dataSources: string[];
  } | null>(null);

  // 已初始化系统不可访问此页 → 重定向
  useEffect(() => {
    // [API] GET /api/v1/auth/status — 检查是否已初始化
    authApi.getStatus().then((status) => {
      if (status.initialized) {
        navigate('/login', { replace: true });
      }
    });
  }, []);

  // Step1 → Step2
  const handleStep1Next = async () => {
    const values = await adminForm.validateFields();
    if (values.password !== values.confirmPassword) {
      message.error('两次密码不一致');
      return;
    }
    setCurrentStep(1);
  };

  // Step2 → Step3：汇总预览
  const handleStep2Next = async () => {
    const values = await configForm.validateFields();
    setSummary({
      username: adminForm.getFieldValue('username'),
      deepseekKey: values.deepseekApiKey.slice(0, 8) + '****',
      tradeMode: values.tradeMode,
      dataSources: [
        values.datasourceMiniqmt && 'miniQMT',
        values.datasourceAkshare && 'AKShare',
      ].filter(Boolean) as string[],
    });
    setCurrentStep(2);
  };

  // Step3: 提交初始化
  const handleLaunch = async () => {
    setSubmitting(true);
    try {
      const adminValues = adminForm.getFieldsValue();
      const configValues = configForm.getFieldsValue();
      // [API] POST /api/v1/auth/setup — 创建管理员
      await authApi.setup({
        username: adminValues.username,
        password: adminValues.password,
      });
      // [API] POST /api/v1/system/config — 保存核心配置
      await systemApi.updateConfig({
        llm_deepseek_key: configValues.deepseekApiKey,
        trade_mode: configValues.tradeMode,
        datasource_miniqmt: configValues.datasourceMiniqmt,
        datasource_akshare: configValues.datasourceAkshare,
      });
      setIsCompleted(true);
    } catch (e: any) {
      message.error(e.message || '宇宙启动失败，请重试');
    } finally {
      setSubmitting(false);
    }
  };

        title="宇宙已开启"

        </div>

        ]} />

        {/* Step 1: 创建管理员 */}

          <Form form={adminForm} layout="vertical" style={{ marginTop: 24 }}>
            <Form.Item name="username" label="管理员用户名" rules={[{ required: true, min: 3 }]}>

            </Form.Item>

            </Form.Item>
            <Button type="primary" block onClick={handleStep1Next}>下一步</Button>
          </Form>
```

```typescript
// services/websocketService.ts

// WebSocket实时推送：行情、交易、风控、系统告警

import { getAccessToken, tryRefreshToken } from '@/auth/tokenManager';

type WsMessageType = 'quote' | 'trade' | 'risk' | 'system' | 'market_state';
type WsHandler = (data: any) => void;

class WebSocketService {
  private ws: WebSocket | null = null;
  private handlers: Map<WsMessageType, Set<WsHandler>> = new Map();
  private reconnectTimer: ReturnType<typeof setTimeout> | null = null;
  private heartbeatTimer: ReturnType<typeof setInterval> | null = null;
  private isManualClose = false;

  connect(): void {
    const token = getAccessToken();
    if (!token) return;

    const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    const wsUrl = `${protocol}//${window.location.host}/api/v1/ws?token=${token}`;

    this.ws = new WebSocket(wsUrl);
    this.isManualClose = false;

    this.ws.onopen = () => {
      this.startHeartbeat();
    };

    this.ws.onmessage = (event) => {
      try {
        const msg = JSON.parse(event.data);
        const { type, payload } = msg;
        const handlers = this.handlers.get(type);
        if (handlers) {
          handlers.forEach(handler => handler(payload));
        }
      } catch {
        // 忽略非JSON消息
      }
    };

    this.ws.onclose = () => {
      this.stopHeartbeat();
      if (!this.isManualClose) {
        this.scheduleReconnect();
      }
    };

    this.ws.onerror = () => {
      // onclose 会处理重连
    };
  }

  disconnect(): void {
    this.isManualClose = true;
    this.stopHeartbeat();
    if (this.reconnectTimer) {
      clearTimeout(this.reconnectTimer);
      this.reconnectTimer = null;
    }
    if (this.ws) {
      this.ws.close();
      this.ws = null;
    }
  }

  subscribe(type: WsMessageType, handler: WsHandler): () => void {
    if (!this.handlers.has(type)) {
      this.handlers.set(type, new Set());
    }
    this.handlers.get(type)!.add(handler);

    // 返回取消订阅函数
    return () => {
      this.handlers.get(type)?.delete(handler);
    };
  }

  private startHeartbeat(): void {
    this.heartbeatTimer = setInterval(() => {
      if (this.ws?.readyState === WebSocket.OPEN) {
        this.ws.send(JSON.stringify({ type: 'ping' }));
      }
    }, 30000);       // 30秒心跳
  }

  private stopHeartbeat(): void {
    if (this.heartbeatTimer) {
      clearInterval(this.heartbeatTimer);
      this.heartbeatTimer = null;
    }
  }

  private scheduleReconnect(): void {
    this.reconnectTimer = setTimeout(() => {
      this.connect();
    }, 5000);       // 5秒后重连
  }
}

export const wsService = new WebSocketService();
```

```typescript
  useEffect(() => {
    const handleResize = () => setIsMobile(window.innerWidth < 768);
    window.addEventListener('resize', handleResize);
    return () => window.removeEventListener('resize', handleResize);
  }, []);

  return { isMobile };
}

// utils/deviceDetect.ts — 设备检测与路由跳转
export function detectAndRedirect(): void {
  const isMobile = /Android|iPhone|iPad|iPod/i.test(navigator.userAgent);
  const isMobileRoute = window.location.pathname.startsWith('/m/');

  if (isMobile && !isMobileRoute) {
    // PC页面 → 移动端首页
    window.location.href = '/m/dashboard';
  } else if (!isMobile && isMobileRoute) {
    // 移动端页面 → PC端首页
    window.location.href = '/dashboard';
  }
}
```

### 6.2 ErrorBoundary 组件

```typescript
// components/common/ErrorBoundary.tsx

import { Component, ErrorInfo, ReactNode } from 'react';
import { Button, Result } from 'antd';
import { FRAX_COLORS } from '@/theme/colors';

interface Props {
  children: ReactNode;
  fallback?: ReactNode;
}

interface State {
  hasError: boolean;
  error: Error | null;
}

export class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props);
    this.state = { hasError: false, error: null };
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo): void {
    console.error('[ErrorBoundary]', error, errorInfo);
  }

  render(): ReactNode {
    if (this.state.hasError) {
      if (this.props.fallback) {
        return this.props.fallback;
      }
      return (
        <Result
          status="500"
          title="星象紊乱"
          subTitle="页面出现异常，请尝试刷新"
          extra={
            <Button type="primary" onClick={() => window.location.reload()}>
              重新观测
            </Button>
          }
          style={{
            background: FRAX_COLORS.cosmosBg,
            color: FRAX_COLORS.textPrimary,
          }}
        />
      );
    }
    return this.props.children;
  }
}
```
