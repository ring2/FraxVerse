## 1. 模块概述

### 1.1 职责边界

本模块负责：

- 基于 miniQMT/xtquant SDK 的交易执行与订单生命周期管理 [PRD-T-120]
- 推进式仓位管理（50% + 5% + 补仓三步入场）[PRD-T-122]
- 摊平禁令检查（浮亏禁止加仓）[PRD-T-123]
- 单票最大风险控制（亏损 ≤ 总资金1.5%）[PRD-T-124]
- 独立进程止损监视器（1秒频率，直接执行卖出）[PRD-T-125][PRD-T-134]
- 止损隔离性保障（不读Agent、无绕过通道、绑定为下单前提）[PRD-T-126~T-129]
- 止损冷却期与经验归档（24小时冷却 + 即时归档）[PRD-T-130~T-132]
- 阶梯止盈执行（硬约束，触发即执行）[PRD-T-135]
- 止盈冷却期（12小时）[PRD-T-136]
- 交易模式管理（SIMULATION/PAPER/LIVE 三态 + 单向升级）[PRD-T-137~T-140]
- 账户资产同步（持仓、资产、盈亏定期同步）
- 交易全链路日志

**不负责**：

- 策略信号生成（见 DD-03）
- Agent 讨论与评分（见 DD-04）
- 经验库存储（见 DD-07，本模块仅触发归档事件）
- 风控事件判定（见 DD-06，但止损执行在本模块）
- 市场数据采集（见 DD-02）

### 2.2 交易模式枚举

class TradeMode(str, Enum):
    SIMULATION = "SIMULATION"   # 模拟盘 — xtquant模拟环境
    PAPER = "PAPER"             # 仿真盘 — miniQMT仿真通道
    LIVE = "LIVE"               # 实盘 — miniQMT实盘通道
    # [PRD-T-140] 交易模式三态：SIMULATION/PAPER/LIVE # PRD L2552 强约束

class ConfirmMode(str, Enum):
    ADVISORY = "advisory"       # 建议模式 — V1默认，推送确认后下单
    SEMI_AUTO = "semi_auto"     # 半自动模式 — 止损止盈自动，新开仓需确认
    FULL_AUTO = "full_auto"     # 全自动模式 — 全自动执行

# 单向升级路径：advisory → semi_auto → full_auto
CONFIRM_MODE_UPGRADE_PATH = ["advisory", "semi_auto", "full_auto"]

#### 3.2.2 PUT /api/v1/trade/mode/upgrade — 升级交易模式

**请求体**：

```json
{
  "targetConfirmMode": "semi_auto",
  "password": "user_password"
}
```

**伪代码**：

```
# 幂等性：否（模式升级有副作用）
function upgrade_trade_mode(target_mode: ConfirmMode, password: str) -> TradeMode:
    # [PRD-T-137] 交易模式单向升级：建议→半自动→全自动 # PRD L728 强约束
    # [PRD-T-138] 模式切换需密码确认 # PRD L729 强约束

    lock = acquire_lock("lock:mode_upgrade", ttl=10)
    if not lock then
        [RAISE] Error(40420, "模式切换操作冲突，请稍后重试")
    end if

    try:
        current = [DB] SELECT * FROM trade_mode LIMIT 1

        # 校验密码
        if not bcrypt_verify(password, current.mode_password_hash) then
            [RAISE] Error(40421, "密码确认失败")
        end if

        # 校验单向升级
        current_idx = index_of(CONFIRM_MODE_UPGRADE_PATH, current.confirm_mode)
        target_idx = index_of(CONFIRM_MODE_UPGRADE_PATH, target_mode)
        if target_idx <= current_idx then
            [RAISE] Error(40422, "交易模式只允许单向升级，不可降级")  # PRD L728 强约束
        end if
        if target_idx > current_idx + 1 then
            [RAISE] Error(40423, "交易模式只能逐级升级")
        end if

        # 执行升级
        [DB] UPDATE trade_mode SET
            confirm_mode = target_mode,
            upgraded_at = NOW(),
            updated_at = NOW()

### 4.4 单票最大风险计算

```
# [PRD-T-124] 单票最大亏损=总资金1.5% # PRD L624 强约束
# 幂等性：是
function calculate_position_risk(stock_code: str, add_volume: int,
                                  add_price: Decimal) -> Decimal:
    total_capital = get_total_capital()
    max_loss_allowed = total_capital * 0.015  # PRD L624 强约束：总资金1.5%

    # 当前持仓风险
    position = [DB] SELECT * FROM positions WHERE stock_code = stock_code AND total_volume > 0
    current_risk = Decimal(0)
    if position is not None then
        stop_loss_cond = [DB] SELECT * FROM stop_loss_conditions
                          WHERE stock_code = stock_code AND is_active = TRUE LIMIT 1
        if stop_loss_cond is not None then
            current_loss_per_share = position.cost_price - stop_loss_cond.stop_loss_price
            current_risk = current_loss_per_share * position.total_volume
        end if
    end if

    # 新增仓位风险
    # 新仓位的止损线 = 买入价 × (1 - max_loss_pct)
    # 批次越多，最后一批止损线越窄
    remaining_budget = max_loss_allowed - current_risk
    if remaining_budget <= 0 then
        return max_loss_allowed + 1  # 超限
    end if

    # 计算新增仓位的最大亏损
    new_stop_loss_pct = min(0.05, remaining_budget / (add_price * add_volume))
    new_loss = add_price * new_stop_loss_pct * add_volume

    total_risk = current_risk + new_loss
    return total_risk
end function

# 辅助：计算新仓位的止损价
function calculate_stop_loss(stock_code: str, entry_price: Decimal,
                              volume: int, total_capital: Decimal) -> tuple:
    max_loss_amount = total_capital * 0.015  # PRD L624

    # 查询已有风险
    existing_risk = get_existing_risk(stock_code)
    available_loss = max_loss_amount - existing_risk

    # 最大止损比例 = 可用亏损额 / (入场价 × 数量)
    max_stop_pct = available_loss / (entry_price * volume)
    max_stop_pct = min(max_stop_pct, 0.05)  # 上限5%
    max_stop_pct = max(max_stop_pct, 0.01)  # 下限1%

    stop_loss_price = entry_price * (1 - max_stop_pct)

    return (stop_loss_price, max_stop_pct)
end function

        self.load_conditions_from_db()

        # 3. 订阅止损条件变更通知
        self.redis_sub = subscribe("stop_loss:update")

        # 4. 主循环 — 每1秒检查一次 [PRD-T-134]
        while True:
            try:
                self.check_stop_conditions()
                self.check_profit_conditions()
                self.process_redis_messages()
            except Exception as e:
                [LOG] ERROR "止损监视器检查异常" extra={"error": str(e)}
            end try
            sleep(1)  # PRD L635 强约束：1秒检查频率
        end while
    end function

    # 加载所有活跃止损条件到内存缓存
    function load_conditions_from_db(self):

        for cond in conditions do
            self.conditions_cache[cond.position_id] = {
                "id": cond.id,
                "stock_code": cond.stock_code,
                "stop_loss_price": cond.stop_loss_price,
                "max_loss_amount": cond.max_loss_amount,
                "condition_type": cond.condition_type,

        end for

        profits = [DB] SELECT * FROM stop_profit_conditions WHERE is_active = TRUE
        for cond in profits do
            key = cond.position_id
            if key not in self.profit_conditions_cache then
                self.profit_conditions_cache[key] = []
            end if
            self.profit_conditions_cache[key].append({
                "id": cond.id,
                "stock_code": cond.stock_code,
                "stage": cond.stage,
                "trigger_pct": cond.trigger_pct,
                "sell_pct": cond.sell_pct
            })
        end for

        [LOG] INFO "止损条件已加载到内存" extra={
            "stop_loss_count": len(self.conditions_cache),

    # 处理Redis消息（止损条件变更通知）
    function process_redis_messages(self):
        while self.redis_sub.has_message() do
            msg = self.redis_sub.get_message()
            payload = json_parse(msg.payload)

            if payload.type == "add" then
                # 新增止损条件
                self.conditions_cache[payload.position_id] = payload.condition
                [LOG] INFO "止损条件已添加到缓存" extra={
                    "position_id": payload.position_id,
                    "stock_code": payload.condition.stock_code
                }
            elif payload.type == "remove" then
                # 移除止损条件（已触发或已平仓）
                del self.conditions_cache[payload.position_id]
                [LOG] INFO "止损条件已从缓存移除" extra={
                    "position_id": payload.position_id
                }
            elif payload.type == "update" then
                # 更新止损条件（如跟踪止损价格上移）
                self.conditions_cache[payload.position_id] = payload.condition
            end if
        end while
    end function

    # 核心检查函数 — 每秒执行一次
    function check_stop_conditions(self):
        # [PRD-T-133] 内存缓存止损条件，盘中只读Redis价格 # PRD L652 强约束
        for position_id, cond in self.conditions_cache.items():
            # 只从Redis读取实时价格，不读数据库，不读Agent结果
            # [PRD-T-126] 止损进程不读Agent分析结果 # PRD L640 强约束
            current_price = [REDIS] GET "price:realtime:{cond.stock_code}"

            if current_price is None then
                continue  # 价格不可用，跳过本次检查
            end if

            # 止损触发判定
            triggered = False

            if cond.condition_type == "fixed_price" then
                if current_price <= cond.stop_loss_price then
                    triggered = True
                end if
            elif cond.condition_type == "trailing" then
                # 跟踪止损：更新最高价
                if current_price > cond.trigger_price then
                    cond.trigger_price = current_price
                    cond.stop_loss_price = current_price * (1 - cond.trailing_pct)
                elif current_price <= cond.stop_loss_price then
                    triggered = True
                end if
            elif cond.condition_type == "max_loss" then
                position = [DB] SELECT * FROM positions WHERE id = position_id  # 仅查持仓
                current_loss = (position.cost_price - current_price) * position.total_volume
                if current_loss >= cond.max_loss_amount then
                    triggered = True
                end if
            end if

            if triggered then
                self.execute_stop_loss(cond, current_price)
            end if
        end for
    end function

function check_profit_conditions(self):
    for position_id, conditions in self.profit_conditions_cache.items():
        if len(conditions) == 0 then
            continue
        end if

        stock_code = conditions[0].stock_code
        current_price = [REDIS] GET "price:realtime:{stock_code}"
        if current_price is None then
            continue
        end if

        position = [DB] SELECT * FROM positions WHERE id = position_id
        if position is None or position.total_volume <= 0 then
            continue
        end if

        profit_pct = (current_price - position.cost_price) / position.cost_price

        for cond in conditions:
            if not cond.is_active then
                continue
            end if

            triggered = False
            sell_volume = 0

            if cond.stage == "first_take" then
                # 第一阶：+8% 卖50%

                    triggered = True
                    sell_volume = floor(position.total_volume * cond.sell_pct)

            elif cond.stage == "second_take" then

                if profit_pct >= cond.trigger_pct then
                    triggered = True
                    sell_volume = floor(position.total_volume * cond.sell_pct)
                end if
            elif cond.stage == "trailing" then
                # 第三阶：MA5移动止盈
                ma5 = get_ma5(stock_code)
                if current_price < ma5 then
                    triggered = True
                    sell_volume = position.available_volume  # 全部卖出
                end if
            end if

            if triggered then
                self.execute_stop_profit(cond, current_price, sell_volume, position)
                break  # 每次只处理一个止盈条件
            end if
        end for
    end for

            TODAY(), NOW()
        )
    else
        # 更新持仓（加仓）
        new_total = position.total_volume + order.filled_volume
        new_cost = (position.cost_price * position.total_volume +
                    order.filled_amount) / new_total

        [DB] UPDATE positions SET
            total_volume = new_total,
            cost_price = new_cost,
            market_value = new_cost * new_total,
            last_trade_at = NOW(),
            updated_at = NOW()
        WHERE id = position.id
    end if
end function

function update_position_on_sell(order: TradeOrder):
    position = [DB] SELECT * FROM positions WHERE stock_code = order.stock_code

    if position is None then
        [LOG] ERROR "卖出但无持仓记录" extra={"stock_code": order.stock_code}
        return
    end if

    new_total = position.total_volume - order.filled_volume

    if new_total <= 0 then
        # 清仓
        [DB] UPDATE positions SET
            total_volume = 0,
            available_volume = 0,
            market_value = 0,
            unrealized_pnl = 0,
            unrealized_pnl_pct = 0,
            batch_stage = 'none',
            updated_at = NOW()
        WHERE id = position.id

        # 取消该票所有活跃止损/止盈条件
        deactivate_all_conditions(position.id, order.stock_code)
    else
        # 部分卖出
        [DB] UPDATE positions SET
            total_volume = new_total,
            available_volume = max(0, position.available_volume - order.filled_volume),
            market_value = new_total * get_realtime_price(order.stock_code),
            last_trade_at = NOW(),
            updated_at = NOW()
        WHERE id = position.id
    end if
end function

    # 1. 标记交易通道不可用
    [REDIS] SET "xtquant:status" = "disconnected" WITH TTL 300

    # 2. 暂停所有新下单
    # 已有止损监视器继续运行（使用独立xtquant连接）

    retry_count = 0
    while retry_count < 10:
        try:
            xtquant_reconnect()
            [REDIS] DEL "xtquant:status"
            [LOG] INFO "miniQMT重连成功"
            return

            retry_count += 1
            sleep(min(2 ** retry_count, 60))  # 指数退避，最大60秒

    end while

    [MQ] PUBLISH channel = "system:alert" message = {
